## 五、Phase 3：ChromaDB + BGE-M3 跑通

> ⏱ 预计：2-3 天
> 🎯 目标：理解向量数据库的核心概念，能用 ChromaDB 存文本、查文本
> 📚 学什么：Embedding 是什么、ChromaDB 基本操作

### 5.1 学习内容

| # | 知识点 | 学到什么程度 |
|---|--------|-------------|
| 1 | Embedding 的直观理解 | 把文字变成一串数字，语义相似 → 数字距离近 |
| 2 | 向量数据库 vs 传统数据库 | 传统数据库用关键词精确匹配，向量数据库用语义模糊匹配 |
| 3 | ChromaDB 核心 API | `create_collection`, `add`, `query` 三个方法 |
| 4 | BGE-M3 模型 | 1024 维向量，中英文都支持 |

### 5.2 Embedding 的直观理解（用类比，不用公式）

```
传统数据库（关键词匹配）：
  你搜"面膜推荐" → 只返回包含"面膜"或"推荐"的文档
  你搜"保湿护肤品分享" → 搜不到上面那条，尽管语义高度相关

向量数据库（语义匹配）：
  你搜"保湿护肤品分享" → 能匹配到"面膜推荐"
  因为两句话的"向量"在数学空间里距离很近

向量就是给每段文字在数学空间里定位一个坐标，
意思相近的句子，坐标就靠近。
```

**任务 3-B：ChromaDB 存取查**

创建 `code/phase3/test_chromadb.py`：

```python
"""Phase 3-B：ChromaDB 增删改查操作"""
import chromadb
from chromadb.utils import embedding_functions

# 1. 加载 Embedding 模型
embed_fn = embedding_functions.SentenceTransformerEmbeddingFunction(
    model_name="BAAI/bge-m3"
)

# 2. 创建 ChromaDB 客户端（持久化到本地目录）
client = chromadb.PersistentClient(path="./phase3_test_db")

# 3. 创建一个 Collection（类似 MySQL 的 Table）
collection = client.get_or_create_collection(
    name="test_collection",
    embedding_function=embed_fn
)

# 4. 插入数据（相当于 SQL INSERT）
print("插入测试数据...")
collection.add(
    documents=[
        "广告应当具有可识别性，能够使消费者辨明其为广告。",
        "大众传播媒介不得以新闻报道形式变相发布广告。",
        "通过大众传播媒介发布的广告应当显著标明'广告'。"
    ],
    metadatas=[
        {"source": "广告法第十四条", "type": "regulation"},
        {"source": "广告法第十四条", "type": "regulation"},
        {"source": "广告法第十四条", "type": "regulation"}
    ],
    ids=["law_001", "law_002", "law_003"]
)
print(f"✅ 插入完成，共 {collection.count()} 条\n")

# 5. 语义检索（相当于 SQL SELECT ... WHERE ... LIKE）
query_text = "新闻里能不能偷偷插广告？"

print(f"查询文本：「{query_text}」")
print("检索最相似的 2 条结果：\n")

results = collection.query(
    query_texts=[query_text],
    n_results=2
)

# 打印检索结果
for i, (doc_id, doc, meta, distance) in enumerate(zip(
    results['ids'][0],
    results['documents'][0],
    results['metadatas'][0],
    results['distances'][0]
)):
    print(f"第{i+1}名 | 相似度距离: {distance:.4f} | 来源: {meta['source']}")
    print(f"       | {doc}")
    print()
```

## 六、Phase 4：RAG 核心链路串联 ⭐

> ⏱ 预计：2-3 天
> 🎯 目标：把 Phase 2 的 LLM + Phase 3 的向量检索串起来，跑通第一条完整的 RAG 链路
> 🎉 **这是整个项目最重要的里程碑！**

### 6.1 学习内容

| # | 知识点 | 学到什么程度 |
|---|--------|-------------|

"""LLM 输出多层容错解析器"""
import json
import re


def safe_parse_json(llm_output: str) -> dict:
    """
    多层容错解析 LLM 输出的 JSON。
    按优先级尝试 5 种策略，确保从各种格式的输出中提取 JSON。
    """
    # 策略 1：直接解析（理想情况）
    try:
        return json.loads(llm_output)
    except json.JSONDecodeError:
        pass

    # 策略 2：提取 ```json ... ``` 代码块
    match = re.search(r'```json\s*(.*?)\s*```', llm_output, re.DOTALL)
    if match:
        try:
            return json.loads(match.group(1))
        except json.JSONDecodeError:
            pass

    # 策略 3：提取 ``` ... ``` 任意代码块
    match = re.search(r'```\s*(.*?)\s*```', llm_output, re.DOTALL)
    if match:
        try:
            return json.loads(match.group(1))
        except json.JSONDecodeError:
            pass

    # 策略 4：正则匹配第一个 { ... } 
    match = re.search(r'\{[^{}]*\}', llm_output, re.DOTALL)
    if match:
        try:
            return json.loads(match.group(0))
        except json.JSONDecodeError:
            pass

    # 策略 5：兜底
    return {
        "is_ad": None,
        "risk_level": "unknown",
        "reasoning": llm_output,
        "parse_error": True
    }
```

```python
"""RAG 核心链路：检索 → 拼装 Prompt → LLM 推理 → 解析"""
import chromadb
from chromadb.utils import embedding_functions
from openai import OpenAI

from config import (
    DEEPSEEK_API_KEY, DEEPSEEK_BASE_URL, LLM_MODEL,
    CHROMA_DB_PATH, EMBEDDING_MODEL, TOP_K
)
from prompt_templates import build_prompt
from parser import safe_parse_json


class RAGPipeline:
    """RAG 推理管线"""

    def __init__(self):
        # 1. 加载 Embedding 模型
        print("加载 BGE-M3 Embedding 模型...")
        self.embed_fn = embedding_functions.SentenceTransformerEmbeddingFunction(
            model_name=EMBEDDING_MODEL
        )

        # 2. 连接 ChromaDB
        self.chroma_client = chromadb.PersistentClient(path=CHROMA_DB_PATH)

        # 3. 获取知识库 Collections
        self.laws_col = self._get_collection("ad_laws")
        self.cases_col = self._get_collection("ad_cases")

        # 4. 初始化 LLM 客户端
        self.llm = OpenAI(api_key=DEEPSEEK_API_KEY, base_url=DEEPSEEK_BASE_URL)

        print("✅ RAG Pipeline 初始化完成")

    def _get_collection(self, name: str):
        """获取或创建 Collection"""
        return self.chroma_client.get_or_create_collection(
            name=name,
            embedding_function=self.embed_fn
        )

    def retrieve(self, query: str) -> dict:
        """多路召回：从法规库和案例库分别检索"""
        laws_result = self.laws_col.query(
            query_texts=[query],
            n_results=TOP_K
        )
        cases_result = self.cases_col.query(
            query_texts=[query],
            n_results=TOP_K
        )

        return {
            "laws": laws_result['documents'][0] if laws_result['documents'] else [],
            "cases": cases_result['documents'][0] if cases_result['documents'] else []
        }

    def detect(self, text: str, prompt_mode: str = "full") -> dict:
        """完整检测流程"""
        # Step 1: 检索
        knowledge = self.retrieve(text)

        # Step 2: 拼装 Prompt
        prompt = build_prompt(
            text=text,
            laws=knowledge["laws"],
            cases=knowledge["cases"],
            mode=prompt_mode
        )

        # Step 3: LLM 推理
        response = self.llm.chat.completions.create(
            model=LLM_MODEL,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.1  # 低温度确保输出稳定
        )

        raw_output = response.choices[0].message.content

        # Step 4: 解析结果
        result = safe_parse_json(raw_output)

        # 附加元信息
        result["_meta"] = {
            "prompt_tokens": response.usage.prompt_tokens,
            "completion_tokens": response.usage.completion_tokens,
            "retrieved_laws": knowledge["laws"],
            "retrieved_cases": knowledge["cases"],
            "raw_output": raw_output
        }

        return result
```

```python
import sys
sys.path.insert(0, "..")

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from models import AdRequest, AdResponse
from core.rag_pipeline import RAGPipeline

app = FastAPI(title="隐性广告识别API", version="1.0.0")

# CORS（允许 Streamlit 跨域访问）
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

# 初始化 RAG Pipeline（全局单例，只加载一次）
pipeline = RAGPipeline()

@app.get("/")
def root():
    return {"message": "隐性广告识别 API 运行中", "docs": "/docs"}

@app.post("/detect", response_model=AdResponse)
def detect_advertisement(request: AdRequest):
    """检测文本是否包含隐性广告"""
    result = pipeline.detect(request.text, prompt_mode=request.mode)
    meta = result.pop("_meta", {})

    return AdResponse(
        is_ad=result.get("is_ad"),
        ad_type=result.get("ad_type"),
        risk_level=result.get("risk_level", "unknown"),
        violation_type=result.get("violation_type", []),
        confidence=result.get("confidence", 0.0),
        reasoning=result.get("reasoning", ""),
        legal_basis=result.get("legal_basis", []),
        similar_cases=result.get("similar_cases", []),
        retrieved_laws=[law[:100] for law in meta.get("retrieved_laws", [])],
        retrieved_cases=[case[:100] for case in meta.get("retrieved_cases", [])],
        tokens_used=meta.get("prompt_tokens", 0) + meta.get("completion_tokens", 0)
    )
```

```python
"""消融实验执行脚本"""
import sys
sys.path.insert(0, "..")

import pandas as pd
import time
from core.rag_pipeline import RAGPipeline
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

# 加载测试集
df = pd.read_csv("test_dataset.csv")
print(f"加载测试集：{len(df)} 条")

# 初始化 Pipeline
pipeline = RAGPipeline()

# 实验配置
experiments = {
    "A_裸LLM": {"mode": "simple"},
    "B_RAG": {"mode": "rag_only"},
    "C_PE": {"mode": "cot"},      # 需在 prompt_templates.py 中补充
    "D_RAG+PE": {"mode": "full"},
}

results = []

for exp_name, config in experiments.items():
    print(f"\n{'='*50}")
    print(f"运行实验：{exp_name}")
    print(f"{'='*50}")

    predictions = []
    start_time = time.time()

    for idx, row in df.iterrows():
        result = pipeline.detect(row["text"], prompt_mode=config["mode"])
        predictions.append(result)
        if (idx + 1) % 20 == 0:
            print(f"  进度：{idx+1}/{len(df)}")

    elapsed = time.time() - start_time

    # 提取判定结果
    pred_is_ad = [p.get("is_ad", False) for p in predictions]
    pred_risk = [p.get("risk_level", "unknown") for p in predictions]

    # 计算指标（与标注对比）

    true_risk = df["label_risk_level"].tolist()

    # 跳过解析失败的样本
    valid_indices = [i for i, p in enumerate(pred_is_ad) if p is not None]

    acc = accuracy_score(
        [true_is_ad[i] for i in valid_indices],
        [pred_is_ad[i] for i in valid_indices]
    )
    prec = precision_score(
        [true_is_ad[i] for i in valid_indices],
        [pred_is_ad[i] for i in valid_indices]
    )
    rec = recall_score(
        [true_is_ad[i] for i in valid_indices],
        [pred_is_ad[i] for i in valid_indices]
    )
    f1 = f1_score(
        [true_is_ad[i] for i in valid_indices],
        [pred_is_ad[i] for i in valid_indices]
    )

    results.append({
        "实验组": exp_name,
        "准确率": f"{acc:.2%}",
        "精确率": f"{prec:.2%}",
        "召回率": f"{rec:.2%}",
        "F1分数": f"{f1:.4f}",
        "有效样本": len(valid_indices),
        "解析失败": len(df) - len(valid_indices),
        "总耗时(秒)": f"{elapsed:.1f}",
        "平均每条(秒)": f"{elapsed/len(df):.2f}",
    })

    # 保存该组详细结果
    pd.DataFrame(predictions).to_csv(f"results/{exp_name}_details.csv", index=False)

# 汇总表
results_df = pd.DataFrame(results)
print("\n\n" + "="*60)
print("消融实验结果汇总")
print("="*60)
